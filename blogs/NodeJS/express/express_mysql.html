数据库准备

创建一个新数据库，之后创建一个表

create table category
(
    create_time datetime    null,
    status      tinyint     null,
    category    varchar(30) not null,
    id          int auto_increment,
    constraint category_pk
        primary key (id)
);

数据库连接

在根目录下创建文件夹util存放工具文件，util下新建文件dbconfig.mjs,这里写连接数据库的配置

const mysql = require('mysql2');

const pool = mysql.createPool({//建立连接池，连接数据库,连接池只创建一次
        host: "localhost",
        port: '3306',
        user: 'root',
        password: 'xxx',
        database: 'fornodejs'
});

module.exports = (sql, sqlArr, callBack) => {
        pool.getConnection((err, conn) => {//回调函数
                console.log('111');
                if(err){
                        console.log('连接失败');
                        return;
                }
                //事件驱动回调
                conn.query(sql, sqlArr, callBack);
                //释放连接
                conn.release();
        })
}

之后转到index.js进行测试

写到这，发现query是异步任务，release是同步任务，感觉会有释放了连接但是查询还没结束的情况发生，于是对代码进行了修改

// dbconfig.js
const mysql = require('mysql2');

const pool = mysql.createPool({//建立连接池，连接数据库,连接池只创建一次
        host: "localhost",
        port: '3306',
        user: 'root',
        password: 'xxx',
        database: 'fornodejs'
});

module.exports = (sql, sqlArr, callBack) => {
        pool.getConnection((err, conn) => {//回调函数
                if(err){
                        console.log('连接失败');
                        return;
                }
                //事件驱动回调
                conn.query(sql, sqlArr, (err, result) => {
                        if(err){
                                callBack(err);
                        }else {
                                callBack(null, result);
                        }
                        
                        conn.release();//这是同步任务，放在query异步任务里，防止提前释放
                        });//查询函数
                //释放连接
        })
}



// cateController.js
const dbConfig = require('../util/dbconfig');

const getCate = (req, res) => {
    let sql = 'select * from category';
    let sqlArr = [];

    dbConfig(sql, sqlArr, (err, result) => {
        if(err){
            console.log('连接出错')
        }
        else{
            res.send({
                list: result
            })
        }
    });
}

module.exports = getCate;
